// --- Universal Smart Column Mapper --- function normalizeRow(rowObj, keys) { const normalized = {}; for (const [k, v] of Object.entries(rowObj)) { normalized[k.toLowerCase().replace(/[^a-z0-9]/g, '')] = v; } for (const key of keys) { const cleanKey = key.toLowerCase().replace(/[^a-z0-9]/g, ''); if (normalized[cleanKey] !== undefined && normalized[cleanKey] !== '') { return normalized[cleanKey]; } } return ''; } // Converts gviz JSON to an array of objects even if labels are empty function parseGvizTable(json) { if (!json || !json.table || !json.table.rows) return []; const cols = json.table.cols; let headers = cols.map(c => (c && c.label ? c.label.trim() : '')); let rows = json.table.rows; // If headers are empty or default column letters (A, B, C), check if row 0 holds the actual headers const hasLabels = headers.some(h => h.length > 1); if (!hasLabels && rows.length > 0) { const firstRowVals = rows[0].c.map(cell => (cell && cell.v !== null ? String(cell.v).trim() : '')); if (firstRowVals.some(v => isNaN(v) && v.length > 0)) { headers = firstRowVals; rows = rows.slice(1); } } return rows.map((r, rowIdx) => { const obj = { _rowIdx: rowIdx }; if (r.c) { r.c.forEach((cell, colIdx) => { const val = (cell && cell.v !== null && cell.v !== undefined) ? cell.v : ''; const headerKey = headers[colIdx] || `col_${colIdx}`; obj[headerKey] = val; obj[`col_${colIdx}`] = val; // Always store by position for fallback }); } return obj; }); } // --- Dynamic Loaders --- window.onBoxesLoaded = function(json) { try { const parsed = parseGvizTable(json); boxes = parsed.map((r, idx) => { const code = normalizeRow(r, ['code', 'giftboxcode', 'boxcode', 'itemcode']) || r.col_0 || `BOX-${idx+1}`; const name = normalizeRow(r, ['giftboxname', 'boxname', 'name', 'box', 'itemname']) || r.col_1 || `Gift Box ${idx+1}`; const rawPrice = normalizeRow(r, ['baseprice', 'price', 'totalprice', 'rate', 'cost']) || r.col_2 || 0; const rawMax = normalizeRow(r, ['maxitem', 'maxitems', 'capacity', 'limit', 'slots', 'max']) || r.col_3 || 5; return { code: String(code).trim(), name: String(name).trim(), price: parseFloat(String(rawPrice).replace(/[^0-9.]/g, '')) || 0, max: parseInt(String(rawMax).replace(/[^0-9]/g, ''), 10) || 5 }; }).filter(b => b.name && b.name !== ''); // Proceed to load Products sheet loadSheetJSONP("Products", "onProductsLoaded"); } catch (e) { console.error("Error parsing Boxes sheet:", e); } }; window.onProductsLoaded = function(json) { try { const parsed = parseGvizTable(json); products = parsed.map((r, idx) => { const cat = normalizeRow(r, ['category', 'cat', 'group', 'type']) || r.col_0 || 'General'; const code = normalizeRow(r, ['code', 'itemcode', 'productcode', 'id', 'sku']) || r.col_1 || `P-${idx+1}`; const name = normalizeRow(r, ['productname', 'name', 'itemname', 'description', 'product']) || r.col_2 || `Item ${idx+1}`; const rawPrice = normalizeRow(r, ['totalprice', 'price', 'mrp', 'rate', 'netprice']) || r.col_5 || r.col_4 || r.col_3 || 0; return { category: String(cat).trim(), code: String(code).trim(), name: String(name).trim(), price: parseFloat(String(rawPrice).replace(/[^0-9.]/g, '')) || 0 }; }).filter(p => p.name && p.name !== '' && !p.name.toLowerCase().includes('product name')); // Load 3rd sheet for Delivery (falls back gracefully if not found) loadSheetJSONP("Delivery", "onDeliveryLoaded"); } catch (e) { console.error("Error parsing Products sheet:", e); } }; window.onDeliveryLoaded = function(json) { try { if (json && json.table && json.table.rows) { const parsed = parseGvizTable(json); const modes = new Set(); const locations = new Set(); parsed.forEach(r => { for (const [k, val] of Object.entries(r)) { if (!val || typeof val !== 'string') continue; const cleanK = k.toLowerCase(); if (cleanK.includes('mode') || cleanK.includes('delivery')) modes.add(val.trim()); else if (cleanK.includes('location') || cleanK.includes('store') || cleanK.includes('branch')) locations.add(val.trim()); } }); if (modes.size > 0) populateSelect("delivery-mode", Array.from(modes)); if (locations.size > 0) populateSelect("delivery-location", Array.from(locations)); } } catch (e) { console.warn("Using default delivery options."); } const syncStatus = document.getElementById('sync-status'); const syncText = document.getElementById('sync-text'); syncStatus.className = "inline-block w-2 h-2 rounded-full bg-emerald-500"; syncText.innerText = `Connected (${products.length} Items, ${boxes.length} Boxes)`; renderBoxes(); renderCategories(); updateUI(); };
